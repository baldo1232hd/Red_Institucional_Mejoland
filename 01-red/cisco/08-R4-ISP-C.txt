! =====================================================================
!  R4 ISP-C (AS 65003)   Router Cisco del rack
!  Router ISR con IOS-XE (interfaces Gi0/0/0 y Gi0/0/1) y servidor DHCP para la red domestica.
!  Antes de aplicar:  enable / write erase / reload
!   Gi0/0/0  -> troncal a SW-NUBE-A Fa0/1 (subinterfaces)
!   Gi0/0/1  -> Casa 1 (WAN del router casero)
! =====================================================================
enable
configure terminal
hostname R4
no ip domain lookup
interface GigabitEthernet0/0/0
 no ip address
 no shutdown
interface GigabitEthernet0/0/0.203
 description MikroTik
 encapsulation dot1Q 203
 ip address 10.0.3.2 255.255.255.252
interface GigabitEthernet0/0/0.214
 description R1 colegio
 encapsulation dot1Q 214
 ip address 10.1.14.1 255.255.255.252
interface GigabitEthernet0/0/0.245
 description R5
 encapsulation dot1Q 245
 ip address 10.0.45.1 255.255.255.252
interface GigabitEthernet0/0/1
 description Casa 1
 ip address 192.168.100.1 255.255.255.0
 no shutdown

router bgp 65003
 bgp router-id 4.4.4.4
 bgp log-neighbor-changes
 redistribute connected
 neighbor 10.0.3.1 remote-as 65001
 neighbor 10.0.3.1 timers 5 15
 neighbor 10.0.45.2 remote-as 65002
 neighbor 10.0.45.2 timers 5 15
 neighbor 10.1.14.2 remote-as 65004
 neighbor 10.1.14.2 timers 5 15
 neighbor 10.1.14.2 default-originate
! ----- DHCP para la casa (el router casero toma su WAN por DHCP) -----
ip dhcp excluded-address 192.168.100.1 192.168.100.9
ip dhcp pool CASA1
 network 192.168.100.0 255.255.255.0
 default-router 192.168.100.1
 dns-server 8.8.8.8 1.1.1.1
end
write memory
