! =====================================================================
!  SEGURIDAD DE ACCESO - comun a todos los equipos Cisco (R1, R3, R4, R5, SW-NUBE-A/B, SWdistri1/2, accesos)
!  Aplicar en cada equipo cambiando solo la linea hostname.
!  Las claves no se incluyen en este archivo: <CLAVE_ADMIN> = usuario SSH, <CLAVE_ENABLE> = enable.
! =====================================================================
enable
configure terminal
hostname R1
ip domain name mejoland.edu.ec
username admin secret <CLAVE_ADMIN>
enable secret <CLAVE_ENABLE>
service password-encryption
security passwords min-length 8
crypto key generate rsa modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
login block-for 120 attempts 3 within 60
ip access-list standard GESTION-SSH
 permit 192.168.99.0 0.0.0.255
 permit 192.168.50.0 0.0.0.255
 deny any log
exit
line vty 0 15
 login local
 transport input ssh
 access-class GESTION-SSH in
 exec-timeout 10 0
exit
line con 0
 login local
 exec-timeout 15 0
 logging synchronous
exit
end
write memory
! Variantes segun el equipo: 'ip domain-name' (con guion) | 'crypto key generate rsa general-keys modulus 2048' | 'line vty 0 4'
