! =====================================================================
!  R3  BORDE + DISTRIBUCION (nucleo colapsado)  AS 65004
!  Nucleo colapsado en router ISR (Gi0/0/0 campus, Gi0/0/1 nube) con NAT estatico hacia el proxy de la DMZ
!  (correo 192.168.60.10 = 10.1.40.13).
!  Antes de aplicar:  enable / write erase / reload
!
!   Gi0/0/1  -> SW-NUBE-B (troncal)   NAT outside / BGP a la nube
!      .213  10.1.3.2/30   MikroTik ISP-A (AS 65001)
!      .235  10.1.35.2/30  R5 ISP-B (AS 65002)
!   Gi0/0/0  -> Switch del campus (troncal con TODAS las VLAN)   NAT inside
!      .100  172.16.100.3/28   transito OSPF con R1
!      .30 .50 .60  HSRP ACTIVO aqui (prioridad 110)
!      .10 .20 .40  HSRP de respaldo (prioridad 90)
!   DHCP: entrega el rango .128 a .254  (R1 entrega .10 a .127)
! =====================================================================
enable
configure terminal
hostname R3
no ip domain lookup

! ===== Lado NUBE: BGP =====
interface GigabitEthernet0/0/1
 description SW-NUBE-B (troncal)
 no ip address
 no shutdown
interface GigabitEthernet0/0/1.213
 description MikroTik ISP-A
 encapsulation dot1Q 213
 ip address 10.1.3.2 255.255.255.252
 ip nat outside
interface GigabitEthernet0/0/1.235
 description R5 (ISP-B)
 encapsulation dot1Q 235
 ip address 10.1.35.2 255.255.255.252
 ip nat outside

! ===== Lado CAMPUS: troncal + subinterfaces (distribucion) =====
interface GigabitEthernet0/0/0
 description Campus (troncal a SWdistri2)
 no ip address
 no shutdown
interface GigabitEthernet0/0/0.100
 description Transito OSPF (R1-R3)
 encapsulation dot1Q 100
 ip address 172.16.100.3 255.255.255.240
 ip ospf priority 0
interface GigabitEthernet0/0/0.10
 description VLAN 10 Administracion
 encapsulation dot1Q 10
 ip address 192.168.10.3 255.255.255.0
 ip nat inside
 standby 10 ip 192.168.10.1
 standby 10 priority 90
 standby 10 preempt
interface GigabitEthernet0/0/0.20
 description VLAN 20 Docentes
 encapsulation dot1Q 20
 ip address 192.168.20.3 255.255.255.0
 ip nat inside
 standby 20 ip 192.168.20.1
 standby 20 priority 90
 standby 20 preempt
interface GigabitEthernet0/0/0.30
 description VLAN 30 Estudiantes
 encapsulation dot1Q 30
 ip address 192.168.30.3 255.255.255.0
 ip nat inside
 standby 30 ip 192.168.30.1
 standby 30 priority 110
 standby 30 preempt
interface GigabitEthernet0/0/0.40
 description VLAN 40 Servidores
 encapsulation dot1Q 40
 ip address 192.168.40.3 255.255.255.0
 ip nat inside
 standby 40 ip 192.168.40.1
 standby 40 priority 90
 standby 40 preempt
interface GigabitEthernet0/0/0.50
 description VLAN 50 Gestion
 encapsulation dot1Q 50
 ip address 192.168.50.3 255.255.255.0
 ip nat inside
 standby 50 ip 192.168.50.1
 standby 50 priority 110
 standby 50 preempt
interface GigabitEthernet0/0/0.60
 description VLAN 60 DMZ Correo
 encapsulation dot1Q 60
 ip address 192.168.60.3 255.255.255.0
 ip nat inside
 standby 60 ip 192.168.60.1
 standby 60 priority 110
 standby 60 preempt

! ===== PAT de salida por cualquiera de los dos proveedores =====
ip access-list standard NAT-INTERNO
 permit 192.168.0.0 0.0.255.255
route-map NAT-ISP-A permit 10
 match ip address NAT-INTERNO
 match interface GigabitEthernet0/0/1.213
route-map NAT-ISP-B permit 10
 match ip address NAT-INTERNO
 match interface GigabitEthernet0/0/1.235
ip nat inside source route-map NAT-ISP-A interface GigabitEthernet0/0/1.213 overload
ip nat inside source route-map NAT-ISP-B interface GigabitEthernet0/0/1.235 overload

! ===== NAT estatico: proxy de la DMZ (correo) publicado como 10.1.40.13 =====
ip nat inside source static 192.168.60.10 10.1.40.13

! ===== Anuncio condicional: solo publica 10.1.40.13 si la DMZ (VLAN 60) esta arriba =====
track 1 ip route 192.168.60.0 255.255.255.0 reachability
 delay down 3 up 10
ip route 10.1.40.13 255.255.255.255 Null0 track 1
ip prefix-list SALIDA-ISP seq 10 permit 10.1.40.13/32

! ===== DHCP (rango .128 a .254) =====
service dhcp
ip dhcp excluded-address 192.168.10.1 192.168.10.127
ip dhcp excluded-address 192.168.20.1 192.168.20.127
ip dhcp excluded-address 192.168.30.1 192.168.30.127
ip dhcp pool VLAN10
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 192.168.40.10 192.168.40.13
 domain-name mejoland.edu.ec
 exit
ip dhcp pool VLAN20
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.40.10 192.168.40.13
 domain-name mejoland.edu.ec
 exit
ip dhcp pool VLAN30
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 192.168.40.10 192.168.40.13
 domain-name mejoland.edu.ec
 exit

! ===== OSPF: anuncia las VLAN + transito y origina el default =====
router ospf 1
 router-id 3.3.3.3
 passive-interface default
 no passive-interface GigabitEthernet0/0/0.100
 network 172.16.100.0 0.0.0.15 area 0
 network 192.168.0.0 0.0.255.255 area 0
 default-information originate metric 20 metric-type 1

! ===== BGP =====
router bgp 65004
 bgp router-id 3.3.3.3
 bgp log-neighbor-changes
 network 10.1.40.13 mask 255.255.255.255
 neighbor 10.1.3.1 remote-as 65001
 neighbor 10.1.3.1 timers 5 15
 neighbor 10.1.3.1 weight 200
 neighbor 10.1.3.1 prefix-list SALIDA-ISP out
 neighbor 10.1.35.1 remote-as 65002
 neighbor 10.1.35.1 timers 5 15
 neighbor 10.1.35.1 prefix-list SALIDA-ISP out
end
write memory
