! =====================================================================
!  R1  BORDE + DISTRIBUCION (nucleo colapsado)  AS 65004
!  Nucleo colapsado en router ISR (Gi0/0/0 campus, Gi0/0/1 nube) con NAT estatico hacia el proxy de la DMZ
!  (damian 192.168.40.13 = 10.1.40.13).
!  Antes de aplicar:  enable / write erase / reload
!
!   Gi0/0/1  -> SW-NUBE-A (troncal)   NAT outside / BGP a la nube
!      .211  10.1.1.2/30   MikroTik ISP-A (AS 65001)
!      .214  10.1.14.2/30  R4 ISP-C (AS 65003)
!   Gi0/0/0  -> Switch del campus (troncal con TODAS las VLAN)   NAT inside
!      .100  172.16.100.1/28   transito OSPF con R3
!      .10 .20 .40  HSRP ACTIVO aqui (prioridad 110)
!      .30 .50 .60  HSRP de respaldo (prioridad 90)
!   DHCP: entrega el rango .10 a .127  (R3 entrega .128 a .254)
! =====================================================================
enable
configure terminal
hostname R1
no ip domain lookup

! ===== Lado NUBE: BGP =====
interface GigabitEthernet0/0/1
 description SW-NUBE-A (troncal)
 no ip address
 no shutdown
interface GigabitEthernet0/0/1.211
 description MikroTik ISP-A
 encapsulation dot1Q 211
 ip address 10.1.1.2 255.255.255.252
 ip nat outside
interface GigabitEthernet0/0/1.214
 description R4 (ISP-C)
 encapsulation dot1Q 214
 ip address 10.1.14.2 255.255.255.252
 ip nat outside

! ===== Lado CAMPUS: troncal + subinterfaces (distribucion) =====
interface GigabitEthernet0/0/0
 description Campus (troncal a SWdistri1)
 no ip address
 no shutdown
interface GigabitEthernet0/0/0.100
 description Transito OSPF (R1-R3)
 encapsulation dot1Q 100
 ip address 172.16.100.1 255.255.255.240
 ip ospf priority 255
interface GigabitEthernet0/0/0.10
 description VLAN 10 Administracion
 encapsulation dot1Q 10
 ip address 192.168.10.2 255.255.255.0
 ip nat inside
 standby 10 ip 192.168.10.1
 standby 10 priority 110
 standby 10 preempt
interface GigabitEthernet0/0/0.20
 description VLAN 20 Docentes
 encapsulation dot1Q 20
 ip address 192.168.20.2 255.255.255.0
 ip nat inside
 standby 20 ip 192.168.20.1
 standby 20 priority 110
 standby 20 preempt
interface GigabitEthernet0/0/0.30
 description VLAN 30 Estudiantes
 encapsulation dot1Q 30
 ip address 192.168.30.2 255.255.255.0
 ip nat inside
 standby 30 ip 192.168.30.1
 standby 30 priority 90
 standby 30 preempt
interface GigabitEthernet0/0/0.40
 description VLAN 40 Servidores
 encapsulation dot1Q 40
 ip address 192.168.40.2 255.255.255.0
 ip nat inside
 standby 40 ip 192.168.40.1
 standby 40 priority 110
 standby 40 preempt
interface GigabitEthernet0/0/0.50
 description VLAN 50 Gestion
 encapsulation dot1Q 50
 ip address 192.168.50.2 255.255.255.0
 ip nat inside
 standby 50 ip 192.168.50.1
 standby 50 priority 90
 standby 50 preempt
interface GigabitEthernet0/0/0.60
 description VLAN 60 DMZ Correo
 encapsulation dot1Q 60
 ip address 192.168.60.2 255.255.255.0
 ip nat inside
 standby 60 ip 192.168.60.1
 standby 60 priority 90
 standby 60 preempt

! ===== PAT de salida por cualquiera de los dos proveedores =====
ip access-list standard NAT-INTERNO
 permit 192.168.0.0 0.0.255.255
route-map NAT-ISP-A permit 10
 match ip address NAT-INTERNO
 match interface GigabitEthernet0/0/1.211
route-map NAT-ISP-C permit 10
 match ip address NAT-INTERNO
 match interface GigabitEthernet0/0/1.214
ip nat inside source route-map NAT-ISP-A interface GigabitEthernet0/0/1.211 overload
ip nat inside source route-map NAT-ISP-C interface GigabitEthernet0/0/1.214 overload

! ===== NAT estatico: damian (192.168.40.13) publicado como 10.1.40.13 =====
ip nat inside source static 192.168.40.13 10.1.40.13

! ===== Anuncio condicional: solo publica 10.1.40.13 si la VLAN de servidores (40) esta arriba =====
track 1 ip route 192.168.40.0 255.255.255.0 reachability
 delay down 3 up 10
ip route 10.1.40.13 255.255.255.255 Null0 track 1
ip prefix-list SALIDA-ISP seq 10 permit 10.1.40.13/32

! ===== DHCP (rango .10 a .127) =====
service dhcp
ip dhcp excluded-address 192.168.10.1 192.168.10.9
ip dhcp excluded-address 192.168.20.1 192.168.20.9
ip dhcp excluded-address 192.168.30.1 192.168.30.9
ip dhcp excluded-address 192.168.10.128 192.168.10.254
ip dhcp excluded-address 192.168.20.128 192.168.20.254
ip dhcp excluded-address 192.168.30.128 192.168.30.254
ip dhcp pool VLAN10
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 192.168.40.10 192.168.40.13
 domain-name mejoland.edu.ec
 exit
ip dhcp pool VLAN20
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.40.10 192.168.40.13
 domain-name mejoland.edu.ec
 exit
ip dhcp pool VLAN30
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 192.168.40.10 192.168.40.13
 domain-name mejoland.edu.ec
 exit

! ===== OSPF: anuncia las VLAN + transito y origina el default =====
router ospf 1
 router-id 1.1.1.1
 passive-interface default
 no passive-interface GigabitEthernet0/0/0.100
 network 172.16.100.0 0.0.0.15 area 0
 network 192.168.0.0 0.0.255.255 area 0
 default-information originate metric 10 metric-type 1

! ===== BGP =====
router bgp 65004
 bgp router-id 1.1.1.1
 bgp log-neighbor-changes
 network 10.1.40.13 mask 255.255.255.255
 neighbor 10.1.1.1 remote-as 65001
 neighbor 10.1.1.1 timers 5 15
 neighbor 10.1.1.1 weight 200
 neighbor 10.1.1.1 prefix-list SALIDA-ISP out
 neighbor 10.1.14.1 remote-as 65003
 neighbor 10.1.14.1 timers 5 15
 neighbor 10.1.14.1 prefix-list SALIDA-ISP out
end
write memory
