vlan 235

vlan 245

 description R5 Gi0/0/0

 description R3 Gi0/0/1

 switchport nonegotiate
 no shutdown

 description SW-NUBE-A Gi1/0/4

 switchport trunk allowed vlan 245
 switchport nonegotiate
 no shutdown
end
